  <script>
    // --- State Variables ---
    let sourceImage = null;
    let audioElement = null;
    let audioBlobUrl = null;
    let audioContext = null;
    let audioSourceNode = null;
    let audioDestNode = null;

    let isRecording = false;
    let animFrameId = null;

    const canvas = document.getElementById('previewCanvas');
    const ctx = canvas.getContext('2d');
    const placeholder = document.getElementById('placeholder');

    // Controls
    const paramDuration = document.getElementById('paramDuration');
    const paramZoom = document.getElementById('paramZoom');
    const paramGlow = document.getElementById('paramGlow');
    const paramParticles = document.getElementById('paramParticles');

    const valDuration = document.getElementById('valDuration');
    const valZoom = document.getElementById('valZoom');
    const valGlow = document.getElementById('valGlow');
    const valParticles = document.getElementById('valParticles');

    const btnRecord = document.getElementById('btnRecord');
    const recordBtnText = document.getElementById('recordBtnText');
    const progressBox = document.getElementById('progressBox');
    const progressFill = document.getElementById('progressFill');
    const progressStatus = document.getElementById('progressStatus');

    const resultCard = document.getElementById('resultCard');
    const outputVideo = document.getElementById('outputVideo');
    const btnDownload = document.getElementById('btnDownload');

    // Update Control Labels
    paramDuration.addEventListener('input', () => valDuration.textContent = paramDuration.value + '초');
    paramZoom.addEventListener('input', () => valZoom.textContent = paramZoom.value + '%');
    paramGlow.addEventListener('input', () => valGlow.textContent = paramGlow.value + '%');
    paramParticles.addEventListener('input', () => valParticles.textContent = paramParticles.value + '개');

    // --- Particle System ---
    let particles = [];
    function initParticles(count, w, h) {
      particles = [];
      const colors = [
        'rgba(255, 200, 100, ', // 따뜻한 주황빛
        'rgba(255, 230, 150, ', // 황금빛
        'rgba(200, 230, 255, ', // 청백색
        'rgba(230, 190, 255, '  // 신비로운 보라빛
      ];
      for (let i = 0; i < count; i++) {
        particles.push({
          x0: Math.random() * w,
          y0: Math.random() * h,
          size: 1.5 + Math.random() * 3.0,
          colorBase: colors[Math.floor(Math.random() * colors.length)],
          swayAmp: 12 + Math.random() * 20,
          swayFreq: Math.floor(Math.random() * 3) + 1,
          phase: Math.random() * Math.PI * 2
        });
      }
    }

    // --- File Input Handlers ---
    document.getElementById('imageInput').addEventListener('change', function(e) {
      const file = e.target.files[0];
      if (!file) return;

      document.getElementById('imageName').textContent = file.name;
      const reader = new FileReader();
      reader.onload = function(evt) {
        const img = new Image();
        img.onload = function() {
          sourceImage = img;
          placeholder.style.display = 'none';
          btnRecord.disabled = false;

          // Adjust canvas aspect ratio to image
          const aspect = img.width / img.height;
          canvas.width = 1280;
          canvas.height = Math.round(1280 / aspect);

          initParticles(parseInt(paramParticles.value), canvas.width, canvas.height);
          startAnimation();
        };
        img.src = evt.target.result;
      };
      reader.readAsDataURL(file);
    });

    document.getElementById('audioInput').addEventListener('change', function(e) {
      const file = e.target.files[0];
      if (!file) return;

      document.getElementById('audioName').textContent = file.name;
      if (audioBlobUrl) URL.revokeObjectURL(audioBlobUrl);

      audioBlobUrl = URL.createObjectURL(file);
      audioElement = new Audio(audioBlobUrl);
      audioElement.loop = true;
    });

    paramParticles.addEventListener('change', () => {
      initParticles(parseInt(paramParticles.value), canvas.width, canvas.height);
    });

    // --- Animation Rendering Pipeline ---
    let startTime = null;

    function renderFrame(timestamp) {
      if (!startTime) startTime = timestamp;
      const durationSec = parseFloat(paramDuration.value);
      const elapsed = (timestamp - startTime) / 1000.0;
      const t = (elapsed % durationSec) / durationSec; // [0.0, 1.0)
      const angle = 2.0 * Math.PI * t;

      const w = canvas.width;
      const h = canvas.height;

      ctx.clearRect(0, 0, w, h);

      if (sourceImage) {
        // 1. Ken Burns Smooth Harmonic Zoom & Pan
        const zoomRange = parseFloat(paramZoom.value) / 100.0;
        const s = 0.5 * (1.0 - Math.cos(angle)); // 0 -> 1 -> 0
        const zoom = 1.0 + zoomRange * s;

        const panX = 12.0 * Math.sin(angle);
        const panY = 8.0 * Math.sin(2.0 * angle);

        const cropW = w / zoom;
        const cropH = h / zoom;
        const cx = w / 2.0 + panX;
        const cy = h / 2.0 + panY;

        ctx.save();
        ctx.drawImage(
          sourceImage,
          (sourceImage.width / w) * (cx - cropW / 2.0),
          (sourceImage.height / h) * (cy - cropH / 2.0),
          (sourceImage.width / w) * cropW,
          (sourceImage.height / h) * cropH,
          0, 0, w, h
        );
        ctx.restore();

        // 2. Ambient Lighting Glow Pulse (Screen / Lighter blend)
        const glowFactor = parseFloat(paramGlow.value) / 100.0;
        if (glowFactor > 0.01) {
          const pulse = glowFactor * 0.5 * (1.0 + Math.sin(angle));
          ctx.save();
          ctx.globalCompositeOperation = 'screen';
          ctx.filter = 'brightness(1.6) contrast(1.8) blur(18px)';
          ctx.globalAlpha = pulse;
          ctx.drawImage(
            sourceImage,
            (sourceImage.width / w) * (cx - cropW / 2.0),
            (sourceImage.height / h) * (cy - cropH / 2.0),
            (sourceImage.width / w) * cropW,
            (sourceImage.height / h) * cropH,
            0, 0, w, h
          );
          ctx.restore();
        }

        // 3. Floating Light Particles (Seamless Wrap & Drift)
        const particleCount = parseInt(paramParticles.value);
        if (particleCount > 0 && particles.length > 0) {
          ctx.save();
          ctx.globalCompositeOperation = 'lighter';
          for (let i = 0; i < Math.min(particleCount, particles.length); i++) {
            const p = particles[i];
            const y = (p.y0 - t * h + h) % h;
            const x = (p.x0 + p.swayAmp * Math.sin(p.swayFreq * angle + p.phase) + w) % w;

            // Fade near edges
            const edgeMargin = 0.12 * h;
            let alpha = 1.0;
            if (y < edgeMargin) alpha = y / edgeMargin;
            else if (y > h - edgeMargin) alpha = (h - y) / edgeMargin;

            const flicker = 0.75 + 0.25 * Math.sin(p.swayFreq * 2 * angle + p.phase);
            const totalAlpha = Math.max(0, Math.min(1, alpha * flicker * 0.8));

            ctx.fillStyle = p.colorBase + totalAlpha + ')';
            ctx.beginPath();
            ctx.arc(x, y, p.size, 0, Math.PI * 2);
            ctx.fill();
          }
          ctx.restore();
        }
      }

      if (!isRecording) {
        animFrameId = requestAnimationFrame(renderFrame);
      }
    }

    function startAnimation() {
      if (animFrameId) cancelAnimationFrame(animFrameId);
      startTime = null;
      animFrameId = requestAnimationFrame(renderFrame);
    }

    // --- MediaRecorder Video Export ---
    btnRecord.addEventListener('click', async function() {
      if (!sourceImage || isRecording) return;

      isRecording = true;
      btnRecord.disabled = true;
      progressBox.style.display = 'flex';
      resultCard.style.display = 'none';

      const durationSec = parseFloat(paramDuration.value);
      const fps = 30;

      // Determine Best Supported MIME Type
      let mimeType = 'video/mp4';
      if (MediaRecorder.isTypeSupported('video/mp4;codecs=avc1')) {
        mimeType = 'video/mp4;codecs=avc1';
      } else if (MediaRecorder.isTypeSupported('video/mp4')) {
        mimeType = 'video/mp4';
      } else if (MediaRecorder.isTypeSupported('video/webm;codecs=vp9')) {
        mimeType = 'video/webm;codecs=vp9';
      } else if (MediaRecorder.isTypeSupported('video/webm')) {
        mimeType = 'video/webm';
      }

      const stream = canvas.captureStream(fps);

      // Audio track integration if available
      let combinedStream = stream;
      let audioNodeToStop = null;
      if (audioElement && audioBlobUrl) {
        try {
          if (!audioContext) {
            audioContext = new (window.AudioContext || window.webkitAudioContext)();
          }
          audioDestNode = audioContext.createMediaStreamDestination();
          audioSourceNode = audioContext.createMediaElementSource(audioElement);
          audioSourceNode.connect(audioDestNode);
          audioSourceNode.connect(audioContext.destination);

          const audioTracks = audioDestNode.stream.getAudioTracks();
          if (audioTracks.length > 0) {
            combinedStream = new MediaStream([
              ...stream.getVideoTracks(),
              audioTracks[0]
            ]);
          }
          audioElement.currentTime = 0;
          audioElement.play();
          audioNodeToStop = audioElement;
        } catch (err) {
          console.warn('오디오 트랙 결합 건너뜀:', err);
        }
      }

      const recorder = new MediaRecorder(combinedStream, {
        mimeType: mimeType,
        videoBitsPerSecond: 8000000 // 8 Mbps high quality
      });

      const chunks = [];
      recorder.ondataavailable = (e) => {
        if (e.data.size > 0) chunks.push(e.data);
      };

      recorder.onstop = () => {
        if (audioNodeToStop) {
          audioNodeToStop.pause();
          audioNodeToStop.currentTime = 0;
        }

        const ext = mimeType.includes('mp4') ? 'mp4' : 'webm';
        const blob = new Blob(chunks, { type: mimeType });
        const videoUrl = URL.createObjectURL(blob);

        outputVideo.src = videoUrl;
        btnDownload.href = videoUrl;
        btnDownload.download = `sts_loop_${Date.now()}.${ext}`;

        resultCard.style.display = 'flex';
        progressBox.style.display = 'none';
        btnRecord.disabled = false;
        recordBtnText.textContent = '루프 동영상 내보내기';
        isRecording = false;

        // Resume live preview animation
        startAnimation();
      };

      // Recording Render Loop
      const totalFrames = Math.round(durationSec * fps);
      let currentFrame = 0;
      recorder.start();

      const renderInterval = 1000.0 / fps;
      const recordStartTime = performance.now();

      function recordStep() {
        if (currentFrame >= totalFrames) {
          recorder.stop();
          return;
        }

        const fakeTimestamp = recordStartTime + (currentFrame / fps) * 1000.0;
        renderFrame(fakeTimestamp);

        currentFrame++;
        const percent = Math.round((currentFrame / totalFrames) * 100);
        progressFill.style.width = percent + '%';
        progressStatus.textContent = `동영상 렌더링 중... ${percent}% (${currentFrame}/${totalFrames} 프레임)`;

        setTimeout(recordStep, renderInterval);
      }

      recordStep();
    });
